= AI0 Video Creator

[plantuml, db-relationship, png]
----
@startuml
' 样式设置
skinparam linetype ortho
skinparam entity {
  BackgroundColor #f9f9f9
  BorderColor #333333
  FontName 微软雅黑, Arial
}
skinparam arrow {
  Color #2c3e50
  LineWidth 1.5
}

' 实体定义
entity "video_projects" as projects {
  * project_uuid : TEXT <<PK>>
  -- 基础字段
  project_name : TEXT
  project_prompt : TEXT
  cover_image_path : TEXT
  create_time : INTEGER
  update_time : INTEGER
  project_status : INTEGER
  -- 扩展字段
  scene_type : TEXT
  scene_config_id : INTEGER <<FK>>
  template_id : INTEGER <<FK>>
  project_path : TEXT
}

entity "scene_config" as scene_config {
  * config_id : INTEGER <<PK, AI>>
  --
  scene_type : TEXT
  script_rules : TEXT
  ai_params : TEXT
  export_config : TEXT
  create_time : DATETIME
  update_time : DATETIME
}

entity "scene_template" as template {
  * template_id : INTEGER <<PK, AI>>
  --
  scene_type : TEXT
  template_name : TEXT
  template_path : TEXT
  template_type : TEXT
  is_default : INTEGER
  create_time : DATETIME
  update_time : DATETIME
}

entity "dialogue_role" as role {
  * role_id : INTEGER <<PK, AI>>
  --
  project_uuid : TEXT <<FK>>
  role_name : TEXT
  role_voice : TEXT
  role_avatar : TEXT
  create_time : DATETIME
  update_time : DATETIME
}

entity "word_detail" as word {
  * word_id : INTEGER <<PK, AI>>
  --
  project_uuid : TEXT <<FK>>
  word : TEXT
  phonetic : TEXT
  paraphrase : TEXT
  example : TEXT
  audio_path : TEXT
  image_path : TEXT
  create_time : DATETIME
  update_time : DATETIME
}

entity "vocabulary" as vocabulary {
  * id : INTEGER <<PK, AI>>
  --
  project_uuid : TEXT <<FK>>
  word : TEXT
  audio_path : TEXT
  index_char : TEXT
  example : TEXT
  image_path : TEXT
  phonetic_symbols : TEXT
  chinese_definition : TEXT
  data : TEXT
  prompt : TEXT
  video_path : TEXT
  ltx23_prompt : TEXT
  t2v_prompt : TEXT
  qwen_image_prompt : TEXT
  category : TEXT
  script : TEXT
  created_at : DATETIME
  updated_at : DATETIME
  status : INTEGER
  chinese : TEXT
}

entity "app_settings" as settings {
  * key : TEXT <<PK>>
  --
  value : TEXT
  updated_at : DATETIME
}

' 关系定义
scene_config ||--o{ projects : 一对多(配置→项目)
template ||--o{ projects : 一对多(模板→项目)
projects ||--o{ role : 一对多(项目→角色)
projects ||--o{ word : 一对多(项目→单词)
projects ||--o{ vocabulary : 一对多(项目→词汇)

@enduml
